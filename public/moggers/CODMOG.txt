        class SoundEngine {
            constructor() { this.ctx = null; }
            init() {
                if (!this.ctx) {
                    this.ctx = new (window.AudioContext || window.webkitAudioContext)();
                }
                if (this.ctx && this.ctx.state === 'suspended') {
                    this.ctx.resume();
                }
            }
            playCoin() {
                if (!this.ctx) return;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(987.77, this.ctx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(1318.51, this.ctx.currentTime + 0.1);
                gain.gain.setValueAtTime(0.12, this.ctx.currentTime);
                gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.1);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(); osc.stop(this.ctx.currentTime + 0.1);
            }
            playJump() {
                if (!this.ctx) return;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(160, this.ctx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(380, this.ctx.currentTime + 0.12);
                gain.gain.setValueAtTime(0.18, this.ctx.currentTime);
                gain.gain.linearRampToValueAtTime(0, this.ctx.currentTime + 0.12);
                osc.connect(gain); gain.connect(this.ctx.destination);
                osc.start(); osc.stop(this.ctx.currentTime + 0.12);
            }
            playStun() {
                if (!this.ctx) return;
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(320, this.ctx.currentTime);
                osc.frequency.linearRampToValueAtTime(120, this.ctx.currentTime + 0.3);

        function spawnChunk(zPos) {
            const randVal = Math.random();

            if (randVal < 0.25) { // Trens tradicionais correndo na mesma direção
                const lane = Math.floor(Math.random() * 3);
                const hasRamp = Math.random() < 0.25;
                spawnTrainObstacle(LANES[lane], zPos, hasRamp, false);
            } else if (randVal < 0.38) {                // Trens correndo na direção oposta (vinda em nossa direção) com frequência moderada
                const lane = Math.floor(Math.random() * 3);
                spawnTrainObstacle(LANES[lane], zPos, false, true);
            } else if (randVal < 0.78) {
                spawnBarrierPattern(zPos);
            } else {
                spawnCoinArcPattern(LANES[Math.floor(Math.random() * 3)], zPos);
            }
        }

        function spawnTrainObstacle(xPos, zPos, hasRamp, isOncoming) {
            const trainGroup = new THREE.Group();
            trainGroup.position.set(xPos, 0, zPos);
            trainGroup.userData = {
                type: 'train',
                hasRamp: hasRamp,
                width: 2.2,
                height: 3.2,
                length: 15.0,
                isOncoming: isOncoming
            };

            const trainLength = 15.0;
            const trainHeight = 3.2;
            const trainWidth = 2.2;

            // Se for trem vindo em nossa direção, mudamos levemente a cor para diferenciá-lo (ex: avermelhado/metálico)
            const trainColor = isOncoming ? 0x991b1b : 0x78350f;
            const body = new THREE.Mesh(
                new THREE.BoxGeometry(trainWidth, trainHeight, trainLength),
                new THREE.MeshStandardMaterial({ color: trainColor, roughness: 0.4, metalness: 0.5 })
            );
            body.position.y = trainHeight / 2;
            body.castShadow = true; body.receiveShadow = true;
            trainGroup.add(body);

            [-0.6, 0.6].forEach(xOff => {
                const light = new THREE.Mesh(
                    new THREE.CylinderGeometry(0.2, 0.2, 0.1, 16),
                    new THREE.MeshBasicMaterial({ color: isOncoming ? 0xef4444 : 0xfef08a })
                );
                light.rotation.x = Math.PI / 2;
                light.position.set(xOff, 1.8, (isOncoming ? -trainLength / 2 - 0.08 : trainLength / 2 + 0.08));
                trainGroup.add(light);
            });

            if (hasRamp && !isOncoming) {
                const rampLength = 3.8;
                const ramp = new THREE.Mesh(
                    new THREE.BoxGeometry(2.1, 0.2, rampLength),
                    new THREE.MeshStandardMaterial({ color: 0xfacc15, roughness: 0.3 })
                );
                ramp.rotation.x = 0.42;
                ramp.position.set(0, 0.75, trainLength / 2 + rampLength / 2 - 0.2);
                trainGroup.add(ramp);
            }

            scene.add(trainGroup);
            obstacles.push(trainGroup);

            if (Math.random() < 0.3 && !isOncoming) {
                for (let z = -4.5; z <= 4.5; z += 2.2) {
                    spawnSingleCoin(xPos, trainHeight + 0.6, zPos + z);
                }
            }
        }

        function createParticles(pos, colorHex, count = 12) {
            const materialKey = String(colorHex);
            if (!particleMaterials.has(materialKey)) {
                particleMaterials.set(materialKey, new THREE.MeshBasicMaterial({ color: colorHex }));
            }
            const material = particleMaterials.get(materialKey);
            const safeCount = Math.min(count, 8);
            while (particles.length > 96) {
                const oldest = particles.shift();
                if (oldest) scene.remove(oldest);
            }
            for (let i = 0; i < safeCount; i++) {
                const p = new THREE.Mesh(particleGeometry, material);
                p.position.copy(pos);
                p.userData = {
                    vel: new THREE.Vector3((Math.random() - 0.5) * 0.2, Math.random() * 0.25 + 0.1, (Math.random() - 0.5) * 0.2),
                    life: 18
                };
                scene.add(p);
                particles.push(p);
            }
        }

        // Hub bridge: the parent d-pad sends one semantic key event per touch.
        // Keep a short per-key guard because some mobile WebViews can emit the
        // same postMessage twice during pointer/touch promotion.
        (function installSurfVT14Bridge() {
            var lastHubKey = '';
            var lastHubKeyAt = 0;
            window.addEventListener('message', function (event) {
                var data = event.data || {};
                if (data.type !== 'SURFVT14_KEY') return;
                var eventType = data.eventType === 'keyup' ? 'keyup' : 'keydown';
                var key = String(data.key || '');
                if (!key) return;
                var now = Date.now();
                if (eventType === 'keydown') {
                    if (key === lastHubKey && now - lastHubKeyAt < 90) return;
                    lastHubKey = key;
                    lastHubKeyAt = now;
                }
                window.dispatchEvent(new KeyboardEvent(eventType, {
                    key: key,
                    code: String(data.code || key),
                    bubbles: true,
                    cancelable: true,
                    repeat: Boolean(data.repeat)
                }));
            });
        })();

        function setupControls() {
            window.addEventListener('keydown', (e) => {
                if (e.key === 'p' || e.key === 'P' || e.key === 'Escape') {
                    if (gameState === 'PLAYING') pauseGame();
                    else if (gameState === 'PAUSED') resumeGame();
                    return;
                }
                if (gameState !== 'PLAYING') return;
                switch (e.key.toLowerCase()) {
                    case 'a': case 'arrowleft': moveLane(-1); break;
                    case 'd': case 'arrowright': moveLane(1); break;
                    case 'w': case 'arrowup': case ' ': jump(); break;
                    case 's': case 'arrowdown': roll(); break;
                }
            });

            let touchStartX = 0, touchStartY = 0;
            let touchMoved = false;

            window.addEventListener('touchstart', (e) => {
                touchStartX = e.touches[0].clientX;
                touchStartY = e.touches[0].clientY;
                touchMoved = false;
            }, { passive: false });

            window.addEventListener('touchmove', (e) => {
                if (gameState !== 'PLAYING') return;

                // Evita que a tela faça scroll no mobile ao tentar deslizar
                if (e.cancelable) e.preventDefault();

                if (touchMoved) return; // Aguarda o próximo toque para deslizar novamente

                const dx = e.touches[0].clientX - touchStartX;
                const dy = e.touches[0].clientY - touchStartY;

                // Reduzido para 25 para uma resposta super rápida
                if (Math.abs(dx) > 25 || Math.abs(dy) > 25) {
                    if (Math.abs(dx) > Math.abs(dy)) {
                        moveLane(dx > 0 ? 1 : -1);
                    } else {
                        if (dy < 0) jump();
                        else roll();
                    }
                    touchMoved = true;
                }
            }, { passive: false });

            document.getElementById('btn-start').addEventListener('click', startGame);
            document.getElementById('btn-restart').addEventListener('click', startGame);
            document.getElementById('btn-pause').addEventListener('click', pauseGame);
            document.getElementById('btn-resume').addEventListener('click', resumeGame);
            document.getElementById('btn-pause-restart').addEventListener('click', startGame);
            if (btnRevive) btnRevive.addEventListener('click', reviveGame);
        }

        function jump() {
            if (isGrounded || activePowerups.sneakers > 0) {
                const force = activePowerups.sneakers > 0 ? JUMP_FORCE * 1.35 : JUMP_FORCE;
                jumpVelocity = force;
                isGrounded = false;
                audio.playJump();
            }
        }

        function roll() {
            if (!isRolling) {
                isRolling = true;
                rollTimer = 45;
                if (!isGrounded) jumpVelocity = -JUMP_FORCE * 1.5;
            }
        }

        function updatePhysics() {
            if (gameState !== 'PLAYING') return;

            // Timer Calculation
            const now = Date.now();
            elapsedTimeMs = now - gameStartTime - totalPausedMs;

            const hh = Math.floor(elapsedTimeMs / 3600000);
            const mm = Math.floor((elapsedTimeMs % 3600000) / 60000);
            const ss = Math.floor((elapsedTimeMs % 60000) / 1000);
            const mms = Math.floor(elapsedTimeMs % 1000);

            hudStopwatch.innerText =
                `${String(hh).padStart(2,'0')}:${String(mm).padStart(2,'0')}:${String(ss).padStart(2,'0')}.${String(mms).padStart(3,'0')}`;

            const totalSecs = elapsedTimeMs / 1000;
            updateLightingAndWeather(totalSecs);

            // Speed Curve
            if (totalSecs < 60) currentSpeed = 0.38;
            else if (totalSecs < 120) currentSpeed = 0.52;
            else if (totalSecs < 600) currentSpeed = 0.75;
            else currentSpeed = 1.10;

            // Stun System Tick
            if (isStunned) {
                stunTimerMs -= 16.6;
                const secsLeft = Math.max(0, Math.ceil(stunTimerMs / 1000));
                stunTimerNum.innerText = secsLeft;

                playerBody.rotation.z = Math.sin(Date.now() * 0.02) * 0.2;
                if (copWarning) copWarning.classList.remove('hidden');

                if (stunTimerMs <= 0) {
                    isStunned = false;
                    playerBody.rotation.z = 0;
                    stunBanner.classList.add('hidden');
                    if (copWarning) copWarning.classList.add('hidden');
                }
            } else {
                playerBody.rotation.z = 0;
                if (copWarning) copWarning.classList.add('hidden');
            }

            let targetInspectorZ = isStunned ? 1.5 : 8.0;
            inspectorGroup.position.z += (targetInspectorZ - inspectorGroup.position.z) * 0.05;
            inspectorGroup.position.x += (playerGroup.position.x - inspectorGroup.position.x) * 0.1;
            inspectorGroup.position.y = isGrounded ? Math.abs(Math.sin(Date.now() * 0.015)) * 0.15 : 0;

            if (isRolling) {
                rollTimer--;
                playerBody.scale.y = 0.5;
                if (rollTimer <= 0) {
                    isRolling = false;
                    playerBody.scale.y = 1;
                }
            } else {
                playerBody.scale.y = 1;
            }

            // Player Jump Physics
            if (!isGrounded) {
                jumpVelocity += GRAVITY;
                playerY += jumpVelocity;
                if (playerY <= 0) {
                    playerY = 0;
                    jumpVelocity = 0;
                    isGrounded = true;
                }
            }

            // Player lateral movement
            playerGroup.position.x += (targetX - playerGroup.position.x) * 0.22;
            playerGroup.position.y = playerY;

            if (isGrounded && !isRolling) {
                const t = Date.now() * 0.015 * (currentSpeed * 2.5);
                leftArm.rotation.x = Math.sin(t) * 0.8;
                rightArm.rotation.x = -Math.sin(t) * 0.8;
                leftLeg.rotation.x = -Math.sin(t) * 0.8;
                rightLeg.rotation.x = Math.sin(t) * 0.8;
            } else if (isRolling) {
                leftArm.rotation.x = -Math.PI / 4;
                rightArm.rotation.x = -Math.PI / 4;
                leftLeg.rotation.x = 0;
                rightLeg.rotation.x = 0;
            } else {
                // Jumping pose
                leftArm.rotation.x = -2.5;
                rightArm.rotation.x = -2.5;
                leftLeg.rotation.x = -0.4;
                rightLeg.rotation.x = 0.2;
            }

            // Track movement
            groundSegments.forEach(seg => {
                seg.position.z += currentSpeed;
                if (seg.position.z > 20) {
                    seg.position.z -= 240;
                    spawnChunk(seg.position.z);
                }
            });

            // Obstacles logic
            for (let i = obstacles.length - 1; i >= 0; i--) {
                const obs = obstacles[i];

                // Se for um trem vindo em nossa direção, ele se move mais rápido em relação ao jogador (velocidade combinada)
                if (obs.userData && obs.userData.type === 'train' && obs.userData.isOncoming) {
                    obs.position.z += currentSpeed * 1.85;
                } else {
                    obs.position.z += currentSpeed;
                }

                // Remover obstáculo quando passar da câmera
                if (obs.position.z > 20) {
                    scene.remove(obs);
                    obstacles.splice(i, 1);
                    continue;
                }

                // Se o obstáculo já foi batido de lado, ignorar novas colisões
                if (obs.userData.hit) continue;

                let obsHalfLength = obs.userData.type === 'train' ? 7.5 : 0.4;
                let frontZExtra = (obs.userData.type === 'train' && obs.userData.hasRamp) ? 3.8 : 0;

                const zFront = obs.position.z + obsHalfLength + frontZExtra;
                const zBack = obs.position.z - obsHalfLength;
                const playerZ = playerGroup.position.z;

                const zOverlap = (playerZ + 0.4 > zBack) && (playerZ - 0.4 < zFront);
                const dx = Math.abs(obs.position.x - playerGroup.position.x);
                const xOverlap = dx < 1.25;

                if (zOverlap && xOverlap) {
                    let collided = false;

                    if (obs.userData.type === 'barrier') {
                        if (!obs.userData.isHigh && playerY > 0.8) {
                            // Pulou a barreira baixa
                        } else if (obs.userData.isHigh && isRolling) {
                            // Rolou por baixo da barreira alta
                        } else {
                            collided = true;
                        }
                    } else if (obs.userData.type === 'train') {
                        const onRampArea = obs.userData.hasRans && (playerZ > obs.position.z + obsHalfLength - 0.5);

                        if (onRampArea && playerY < 0.5 && dx < 0.8) {
                            // Subiu na rampa (apenas no centro)
                            playerY = obs.userData.height;
                            isGrounded = true;
                        } else if (playerY < obs.userData.height - 0.2) {
                            collided = true;
                        }
                    }

                    if (collided) {
                        // Se o dx for grande, significa que foi pego no meio de uma troca de faixa (bateu lateral)
                        const isSideHit = dx >= 0.65;

                        if (!isSideHit) {
                            gameOver();
                            return;
                        } else {
                            // Batida lateral (Tontura/Stun)
                            if (isStunned) {
                                gameOver();
                                return;
                            } else {
                                triggerStun();

                                const obsLane = LANES.findIndex(l => Math.abs(l - obs.position.x) < 0.1);
                                if (obsLane !== -1) {
                                    if (playerGroup.position.x < obs.position.x) {
                                        currentLane = Math.max(0, obsLane - 1);
                                    } else {
                                        currentLane = Math.min(2, obsLane + 1);
                                    }
                                    targetX = LANES[currentLane];
                                }

                                obs.userData.hit = true;
                            }
                        }
                    }
                }
            }

            // Coins logic
            for (let i = coins.length - 1; i >= 0; i--) {
                const coin = coins[i];
                coin.position.z += currentSpeed;
                coin.rotation.z += 0.05;

                const dist = coin.position.distanceTo(playerGroup.position);

                if (activePowerups.magnet > 0 && dist < 8.0) {
                    coin.position.lerp(playerGroup.position, 0.2);
                }

                if (dist < 1.2) {
                    coinsCollected += activePowerups.multiplier > 0 ? 2 : 1;
                    score += 50;
                    audio.playCoin();
                    createParticles(coin.position, 0xfacc15, 8);
                    scene.remove(coin);
                    coins.splice(i, 1);
                    continue;
                }

                if (coin.position.z > 20) {
                    scene.remove(coin);
                    coins.splice(i, 1);
                }
            }

            // Powerups logic
            for (let i = powerups.length - 1; i >= 0; i--) {
                const pw = powerups[i];
                pw.position.z += currentSpeed;
                pw.rotation.y += 0.04;

                const dist = pw.position.distanceTo(playerGroup.position);
                if (dist < 1.4) {
                    const type = pw.userData.type;
                    activePowerups[type] = 360;
                    if (type === 'jetpack') {
                        playerBody.getObjectByName("jetpack").visible = true;
                    }
                    audio.playPowerup();
                    scene.remove(pw);
                    powerups.splice(i, 1);
                    continue;
                }

                if (pw.position.z > 20) {
                    scene.remove(pw);
                    powerups.splice(i, 1);
                }
            }

            // Update Multiplier Display
            if (multTag) {
                const multVal = activePowerups.multiplier > 0 ? 2 : 1;
                multTag.innerText = `(x${multVal})`;
            }

            updatePowerupTimers();

            // Particles update
            for (let i = particles.length - 1; i >= 0; i--) {
                const part = particles[i];
                part.position.add(part.userData.vel);
                part.userData.life--;
                if (part.userData.life <= 0) { scene.remove(part); particles.splice(i, 1); }
            }

            score += Math.round(currentSpeed * 2);
            updateHUD();

            // Camera Tracking
            let shake = isStunned ? (Math.random() - 0.5) * 0.18 : 0;
            camera.position.x += (playerGroup.position.x * 0.3 - camera.position.x) * 0.1 + shake;
            camera.position.y += (playerGroup.position.y * 0.2 + 3.8 - camera.position.y) * 0.1;
        }

        function triggerStun() {
            isStunned = true;
            stunTimerMs = 5000;
            stunBanner.classList.remove('hidden');
            if (copWarning) copWarning.classList.remove('hidden');
            audio.playStun();
            audio.playWhistle();
            createParticles(playerGroup.position, 0xef4444, 12);
        }

        function updatePowerupTimers() {
            powerupTimersContainer.innerHTML = '';
            Object.keys(activePowerups).forEach(type => {
                if (activePowerups[type] > 0) {
                    activePowerups[type]--;
                    if (activePowerups[type] === 0 && type === 'jetpack') {
                        playerBody.getObjectByName("jetpack").visible = false;
                    }
                    const pct = (activePowerups[type] / 360) * 100;
                    const timerEl = document.createElement('div');
                    timerEl.className = 'w-full hud-glass rounded-xl px-3 py-1 flex items-center gap-3 border-cyan-400/30';
                    timerEl.innerHTML = `
                        <span class="text-[10px] font-black text-cyan-300 w-20 uppercase">${type === 'hoverboard' ? 'PRANCHA' : type}</span>
                        <div class="w-full bg-slate-800 h-2 rounded-full overflow-hidden">
                            <div class="bg-gradient-to-r from-cyan-400 to-amber-400 h-full transition-all duration-75" style="width: ${pct}%"></div>
                        </div>
                    `;
                    powerupTimersContainer.appendChild(timerEl);
                }
            });
        }

        function updateHUD() {
            hudCoins.innerText = coinsCollected.toLocaleString();
            hudScore.innerText = Math.floor(score).toString().padStart(6, '0');
        }

        function startGame() {
            audio.init();

            gameState = 'PLAYING';
            score = 0;
            coinsCollected = 0;
            currentSpeed = 0.38;
            currentLane = 1;
            targetX = LANES[1];
            playerY = 0;
            jumpVelocity = 0;
            isGrounded = true;
            isRolling = false;

            gameStartTime = Date.now();
            totalPausedMs = 0;
            isStunned = false;
            stunTimerMs = 0;
            reviveCost = 10;
            revivesUsed = 0;

            stunBanner.classList.add('hidden');
            if (copWarning) copWarning.classList.add('hidden');

            Object.keys(activePowerups).forEach(k => activePowerups[k] = 0);
            if (playerBody && playerBody.getObjectByName("jetpack")) {
                playerBody.getObjectByName("jetpack").visible = false;
            }

            obstacles.forEach(o => scene.remove(o));
            coins.forEach(c => scene.remove(c));
            powerups.forEach(p => scene.remove(p));
            particles.forEach(p => scene.remove(p));

            obstacles = []; coins = []; powerups = []; particles = [];

            if (playerGroup) playerGroup.position.set(LANES[1], 0, 0);
            if (inspectorGroup) inspectorGroup.position.set(LANES[1], 0, 8.0);

            startScreen.classList.add('hidden');
            pauseScreen.classList.add('hidden');
            gameoverScreen.classList.add('hidden');
            hud.classList.remove('hidden');

            updateHUD();
        }

        function pauseGame() {
            if (gameState !== 'PLAYING') return;
            gameState = 'PAUSED';
            pauseStartTime = Date.now();
            pauseScreen.classList.remove('hidden');
        }

        function resumeGame() {
            if (gameState !== 'PAUSED') return;
            totalPausedMs += (Date.now() - pauseStartTime);
            gameState = 'PLAYING';
            pauseScreen.classList.add('hidden');
        }

        function reviveGame() {
            if (coinsCollected < reviveCost) return;

            coinsCollected -= reviveCost;
            revivesUsed++;
            reviveCost *= 2;

            obstacles.forEach(o => {
                if (Math.abs(o.position.z - playerGroup.position.z) < 30) {
                    scene.remove(o);
                }
            });
            obstacles = obstacles.filter(o => Math.abs(o.position.z - playerGroup.position.z) >= 30);

            isStunned = false;
            stunTimerMs = 0;
            playerY = 0;
            jumpVelocity = 0;
            isGrounded = true;
            playerGroup.position.y = 0;

            stunBanner.classList.add('hidden');
            if (copWarning) copWarning.classList.add('hidden');
            gameoverScreen.classList.add('hidden');
            hud.classList.remove('hidden');

            audio.playPowerup();
            updateHUD();
            gameState = 'PLAYING';
        }

        function gameOver() {
            gameState = 'GAMEOVER';
            audio.playHit();

            if (score > highScore) {
                highScore = Math.floor(score);
                localStorage.setItem('subway_moggers_highscore', highScore);
            }

            document.getElementById('go-time').innerText = hudStopwatch.innerText;
            document.getElementById('go-score').innerText = Math.floor(score).toLocaleString();
            document.getElementById('go-coins').innerText = '🪙' + coinsCollected.toLocaleString();
            document.getElementById('go-highscore').innerText = highScore.toLocaleString();
            if (hudHighscore) hudHighscore.innerText = highScore.toLocaleString();

            if (btnRevive && reviveCostNum) {
                reviveCostNum.innerText = reviveCost;
                if (coinsCollected >= reviveCost) {
                    btnRevive.classList.remove('hidden');
                } else {
                    btnRevive.classList.add('hidden');
                }
            }

            hud.classList.add('hidden');
            gameoverScreen.classList.remove('hidden');
        }

        function onWindowResize() {
            const container = document.getElementById('canvas-container');
            if (!container || !renderer || !camera) return;
            const width = window.innerWidth;
            const height = window.innerHeight;
            camera.aspect = width / height;
            camera.updateProjectionMatrix();
            renderer.setSize(width, height);
        }

        function animate() {
            requestAnimationFrame(animate);
            if (gameState === 'PLAYING') {
                updatePhysics();
            }
            if (renderer && scene && camera) {
                renderer.render(scene, camera);
            }
        }

        // Initialize when window loads completely
        window.addEventListener('load', initEngine);
    </script>
</body>
